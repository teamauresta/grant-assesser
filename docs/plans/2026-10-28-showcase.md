> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Remove decorative coloured strips site-wide and rebuild the home page's motion so every effect shows something real: an aurora hero with a pause button, a scroll-driven "programs sort themselves" story, product previews in How it works, and a cursor spotlight on cards.

```ts
/** `base` blended towards `over` by `amount` (0..1), per sRGB channel, as #rrggbb. */
export function mixHex(base: string, over: string, amount: number): string {
  const ch = (hex: string, i: number) => parseInt(hex.slice(1 + i * 2, 3 + i * 2), 16);
  return '#' + [0, 1, 2]
    .map((i) => Math.round(ch(base, i) + (ch(over, i) - ch(base, i)) * amount).toString(16).padStart(2, '0'))
    .join('');
}
```

```ts
/**
 * The home hero's aurora: a slow field of the Liftoff stops drawn by a WebGL shader
 * (components/home/aurora-canvas.tsx). Shader sources, strength and the remembered pause live here.
 */

/** Strongest the field may paint over --bg, per theme. tests/lib/tokens.test.ts keeps text AA on it. */
export const AURORA_ALPHA = { light: 0.3, dark: 0.18 } as const;

export const AURORA_KEY = 'gn-aurora';

/** True when the person paused the aurora on an earlier visit. Blocked storage reads as playing. */
export function readAuroraPaused(): boolean {
  try { return window.localStorage.getItem(AURORA_KEY) === 'paused'; } catch { return false; }
}

export function writeAuroraPaused(paused: boolean): void {
  try {
    if (paused) window.localStorage.setItem(AURORA_KEY, 'paused');
    else window.localStorage.removeItem(AURORA_KEY);
  } catch { /* blocked storage: the choice lasts for this page only */ }
}

/** #rrggbb as 0..1 channels for a shader uniform; black for anything else. */
export function hexToRgb(hex: string): [number, number, number] {
  const m = /^#([0-9a-f]{6})$/i.exec(hex.trim());
  if (!m) return [0, 0, 0];
  const n = parseInt(m[1], 16);
  return [((n >> 16) & 255) / 255, ((n >> 8) & 255) / 255, (n & 255) / 255];
}

export type AuroraColours = { bg: number[]; stops: number[][]; alpha: number };

/** The current theme's colours, read from the CSS tokens (the theme lives on <html data-theme>). */
export function readAuroraColours(): AuroraColours {
  const root = document.documentElement;
  const style = getComputedStyle(root);
  const read = (name: string) => hexToRgb(style.getPropertyValue(name));
  return {
    bg: read('--bg'),
    stops: ['--lift-1', '--lift-2', '--lift-3'].map(read),
    alpha: AURORA_ALPHA[root.dataset.theme === 'dark' ? 'dark' : 'light'],
  };
}

export const AURORA_VERTEX = `
attribute vec2 aPos;
void main() { gl_Position = vec4(aPos, 0.0, 1.0); }
`;

/**
 * Domain-warped value noise mapped onto the three stops, blended over the page background and
 * strongest to the right (the text column is on the left). Fades out at the top and bottom edges
 * so the hero has no visible boundary. Output alpha never exceeds uAlpha.
 */
export const AURORA_FRAGMENT = `
precision mediump float;
uniform vec2 uRes;
uniform float uTime;
uniform vec2 uPointer;
uniform vec3 uBg;
uniform vec3 uC1;
uniform vec3 uC2;
uniform vec3 uC3;
uniform float uAlpha;

float hash(vec2 p) { return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453); }
float noise(vec2 p) {
  vec2 i = floor(p);
  vec2 f = fract(p);
  vec2 u = f * f * (3.0 - 2.0 * f);
  return mix(mix(hash(i), hash(i + vec2(1.0, 0.0)), u.x), mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x), u.y);
}
float fbm(vec2 p) {
  float v = 0.0;
  float a = 0.5;
  for (int i = 0; i < 5; i++) { v += a * noise(p); p = p * 2.02 + vec2(1.7, 9.2); a *= 0.5; }
  return v;
}

void main() {
  vec2 uv = gl_FragCoord.xy / uRes;
  vec2 p = vec2(uv.x * uRes.x / uRes.y, uv.y) * 1.6;
  float t = uTime * 0.04;
  vec2 d = uv - uPointer;
  float pull = exp(-dot(d, d) * 18.0) * 0.35;
  vec2 q = vec2(fbm(p + t), fbm(p + vec2(5.2, 1.3) - t));
  vec2 r = vec2(fbm(p + 3.0 * q + vec2(1.7, 9.2) + t * 1.3 + pull), fbm(p + 3.0 * q + vec2(8.3, 2.8) - t));
  float f = fbm(p + 2.5 * r);
  vec3 col = mix(uC1, uC2, smoothstep(0.2, 0.6, f));
  col = mix(col, uC3, smoothstep(0.5, 0.85, r.x));
  float reach = smoothstep(0.1, 0.85, uv.x) * (0.55 + 0.45 * f);
  reach *= smoothstep(0.0, 0.2, uv.y) * smoothstep(1.0, 0.8, uv.y);
  gl_FragColor = vec4(mix(uBg, col, uAlpha * clamp(reach, 0.0, 1.0)), 1.0);
}
`;
```

```tsx
'use client';

import { useEffect, useRef } from 'react';
import { AURORA_FRAGMENT, AURORA_VERTEX, readAuroraColours } from '@/lib/aurora';

/** The field is soft, so it renders at half resolution, and never above 1.5 device pixels per CSS pixel. */
const SCALE = 0.5;
/** A pleasing moment of the flow for the first frame (and the only one under reduced motion). */
const START_TIME = 12;

function compile(gl: WebGLRenderingContext, type: number, source: string) {
  const shader = gl.createShader(type);
  if (!shader) return null;
  gl.shaderSource(shader, source);
  gl.compileShader(shader);
  return gl.getShaderParameter(shader, gl.COMPILE_STATUS) ? shader : null;
}

function link(gl: WebGLRenderingContext) {
  const vs = compile(gl, gl.VERTEX_SHADER, AURORA_VERTEX);
  const fs = compile(gl, gl.FRAGMENT_SHADER, AURORA_FRAGMENT);
  const program = gl.createProgram();
  if (!vs || !fs || !program) return null;
  gl.attachShader(program, vs);
  gl.attachShader(program, fs);
  gl.linkProgram(program);
  return gl.getProgramParameter(program, gl.LINK_STATUS) ? program : null;
}

/**
 * Draws the hero aurora. Runs only while `playing`, on screen and in a visible tab; otherwise it
 * keeps its last frame. Follows the pointer gently, re-reads the colours when the theme changes,
 * and writes a frame count to `data-frame` about twice a second (the end-to-end tests read it).
 */
export default function AuroraCanvas({ playing, onUnsupported }: { playing: boolean; onUnsupported: () => void }) {
  const ref = useRef<HTMLCanvasElement>(null);
  const playingRef = useRef(playing);
  const wakeRef = useRef<() => void>(() => {});

  useEffect(() => {
    playingRef.current = playing;
    wakeRef.current();
  }, [playing]);

  useEffect(() => {
    const canvas = ref.current;
    const gl = canvas?.getContext('webgl', { antialias: false, alpha: false, powerPreference: 'low-power' });
    const program = gl && link(gl);
    if (!canvas || !gl || !program) { onUnsupported(); return; }

    gl.useProgram(program);

      const { bg, stops, alpha } = readAuroraColours();
      gl.uniform3fv(u('uBg'), bg);
      gl.uniform3fv(u('uC1'), stops[0]);
      gl.uniform3fv(u('uC2'), stops[1]);
      gl.uniform3fv(u('uC3'), stops[2]);
      gl.uniform1f(u('uAlpha'), alpha);
    };
    const resize = () => {
      const { width, height } = canvas.getBoundingClientRect();
      const k = Math.min(window.devicePixelRatio || 1, 1.5) * SCALE;
      canvas.width = Math.max(1, Math.round(width * k));
      canvas.height = Math.max(1, Math.round(height * k));
      gl.viewport(0, 0, canvas.width, canvas.height);
    };

    let raf = 0;
    let last = 0;
    let time = START_TIME;
    let frames = 0;
    let inView = true;
    let visible = !document.hidden;
    const pointer = { x: 0.75, y: 0.5 };
    const target = { x: 0.75, y: 0.5 };

    const draw = () => {
      gl.uniform2f(uRes, canvas.width, canvas.height);
      gl.uniform1f(uTime, time);
      gl.uniform2f(uPointer, pointer.x, pointer.y);
      gl.drawArrays(gl.TRIANGLES, 0, 3);
    };
    const running = () => playingRef.current && inView && visible;
    const tick = (t: number) => {
      raf = 0;
      time += last ? Math.min(0.05, (t - last) / 1000) : 0;
      last = t;
      pointer.x += (target.x - pointer.x) * 0.04;
      pointer.y += (target.y - pointer.y) * 0.04;
      draw();
      frames += 1;
      if (frames % 30 === 0) canvas.dataset.frame = String(frames);
      if (running()) raf = requestAnimationFrame(tick);
      else last = 0;
    };
    const wake = () => { if (!raf && running()) raf = requestAnimationFrame(tick); };
    wakeRef.current = wake;

    setColours();
    resize();
    draw();
    canvas.dataset.frame = '0';
    canvas.style.opacity = '1';
    wake();

    const io = new IntersectionObserver(([entry]) => { inView = entry.isIntersecting; wake(); });
    io.observe(canvas);
    const ro = new ResizeObserver(() => { resize(); draw(); });
    ro.observe(canvas);
    const theme = new MutationObserver(() => { setColours(); draw(); });
    theme.observe(document.documentElement, { attributes: true, attributeFilter: ['data-theme'] });
    const onVisibility = () => { visible = !document.hidden; wake(); };
    document.addEventListener('visibilitychange', onVisibility);
    const onPointer = (e: PointerEvent) => {
      const r = canvas.getBoundingClientRect();
      target.x = (e.clientX - r.left) / r.width;
      target.y = 1 - (e.clientY - r.top) / r.height;
    };
    window.addEventListener('pointermove', onPointer, { passive: true });
    const onLost = (e: Event) => { e.preventDefault(); onUnsupported(); };
    canvas.addEventListener('webglcontextlost', onLost);

    return () => {
      cancelAnimationFrame(raf);
      wakeRef.current = () => {};
      io.disconnect();
      ro.disconnect();
      theme.disconnect();
      document.removeEventListener('visibilitychange', onVisibility);
      window.removeEventListener('pointermove', onPointer);
      canvas.removeEventListener('webglcontextlost', onLost);
      gl.getExtension('WEBGL_lose_context')?.loseContext();
    };
  }, [onUnsupported]);

  return <canvas ref={ref} className="size-full opacity-0 transition-opacity duration-700 ease-out" />;
}
```

```ts
 * The example founder in the home page's sorting story: a pre-raise AI product company in Victoria,
 * about 18 months old, with angel money. Fixed answers, presented as an example only.
 */
export const DEMO_PROFILE: Profile = {
  // copy every field of preRaiseVic from tests/fixtures/personas.ts (it spreads bootstrappedVic);
  // lib code must not import from tests/.
};

export type StoryChip = { slug: string; name: string; group: Group };
export type Story = { groups: Record<Group, StoryChip[]>; closed: number };

/** Every program checked against DEMO_PROFILE, best first within each group; closed ones are only counted. */
export function sortStory(programs: Program[], now: Date): Story {
  const groups = Object.fromEntries(GROUP_ORDER.map((g) => [g, [] as StoryChip[]])) as Record<Group, StoryChip[]>;
  const assessed = assessAll(programs, DEMO_PROFILE, now);
  for (const a of assessed) groups[a.group!].push({ slug: a.program.slug, name: a.program.name, group: a.group! });
  return { groups, closed: programs.length - assessed.length };
}

export type Scatter = { x: number; y: number; rotate: number; delay: number };

const round = (n: number) => Math.round(n * 100) / 100;

/** FNV-1a, then mulberry32: a small seeded generator, so a slug always scatters the same way. */
function seeded(text: string) {
  let s = 0x811c9dc5;
  for (let i = 0; i < text.length; i++) s = Math.imul(s ^ text.charCodeAt(i), 0x01000193);
  return () => {
    s = (s + 0x6d2b79f5) | 0;
    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/**
 * Where a chip starts before it sorts: an offset from its sorted place (rem), a tilt (degrees) and
 * a stagger (0..1). Flattened vertically, because the stage is wider than it is tall.
 */
export function chipScatter(slug: string): Scatter {
  const next = seeded(slug);
  const angle = next() * Math.PI * 2;
  const distance = 8 + next() * 14;
  return { x: round(Math.cos(angle) * distance), y: round(Math.sin(angle) * distance * 0.6), rotate: round((next() - 0.5) * 24), delay: round(next() * 0.99) };
}

/** Share of the sort each chip spends moving; the rest of the range staggers their starts. */
export const CHIP_SPAN = 0.55;

/** Scroll progress through the pinned section at which every chip has landed; the rest holds the result. */
export const SORT_END = 0.8;

/** A chip's own progress (0..1, eased out) at overall progress `p`. */
export function chipProgress(p: number, delay: number): number {
  const start = delay * (1 - CHIP_SPAN);
  const t = Math.min(1, Math.max(0, (p - start) / CHIP_SPAN));
  return 1 - (1 - t) ** 3;
}
```

```ts
'use client';

import { useEffect, useRef } from 'react';

/**
 * One pointer listener for a list of `[data-spotlight]` cards: writes the pointer's position on the
 * hovered card as --spot-x/--spot-y, read by the `spotlight` utility in app/globals.css. Mouse and
 * pen only; the CSS hides the highlight on touch and under reduced motion.
 */
export function useSpotlight<T extends HTMLElement>() {
  const ref = useRef<T>(null);
  useEffect(() => {
    const root = ref.current;
    if (!root) return;
    const move = (e: PointerEvent) => {
      if (e.pointerType === 'touch') return;
      const card = (e.target as Element | null)?.closest<HTMLElement>('[data-spotlight]');
      if (!card || !root.contains(card)) return;
      const r = card.getBoundingClientRect();
      card.style.setProperty('--spot-x', `${e.clientX - r.left}px`);
      card.style.setProperty('--spot-y', `${e.clientY - r.top}px`);
    };
    root.addEventListener('pointermove', move);
    return () => root.removeEventListener('pointermove', move);
  }, []);
  return ref;
}
```
