# Liftoff Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Give Grant Navigator an Antigravity-style brand, motion and celebration layer that encourages founders, without hurting speed, accessibility or honesty.

**Architecture:** A gradient token set ("Liftoff") and dark-panel tokens in `app/globals.css`, guarded by a contrast unit test. Motion is split by cost: cheap effects (reveals, hover, press, route and dialog entrances, the floating band) are CSS driven by tiny client hooks; layout animations (pipeline glide, directory reflow, tab indicator) use `motion` through `LazyMotion` with `domMax` loaded asynchronously; the hero particle field (`tsParticles`) and confetti (`canvas-confetti`) load only through `import()`. Celebrations go through one `cheer()` function that shows a toast once per event and fires a burst only when the user allows motion and has Celebrations on.

**Tech Stack:** Next.js 16 static export, React 19, TypeScript 6, Tailwind CSS 4, Zustand 5, Zod 4, `motion` 14.6, `@tsparticles/react` + `@tsparticles/slim` + `@tsparticles/engine` 4.5, `canvas-confetti` 1.9.4 (+ `@types/canvas-confetti` 1.9.0), Vitest 5, Playwright 1.64.

```js
#!/usr/bin/env node
// Gzipped size of the JavaScript the home page references up front (out/index.html), against a
// budget. Chunks loaded later with import() are not referenced there, so they do not count.
import { readFileSync } from 'node:fs';
import { gzipSync } from 'node:zlib';
import { join } from 'node:path';
import { fileURLToPath } from 'node:url';

/** Home first-load JS before Liftoff (256,123 bytes on 27 Oct 2026) plus 40 KiB. */
export const BUDGET_BYTES = 256_123 + 40 * 1024;

/** Script paths (relative to out/) referenced anywhere in the page, without duplicates. */
export function firstLoadScripts(html) {
  return [...new Set([...html.matchAll(/\/grant-assesser\/(_next\/static\/[^"'\s]+?\.js)/g)].map((m) => m[1]))];
}

function main() {
  const out = join(fileURLToPath(new URL('..', import.meta.url)), 'out');
  const scripts = firstLoadScripts(readFileSync(join(out, 'index.html'), 'utf8'));
  const total = scripts.reduce((sum, path) => sum + gzipSync(readFileSync(join(out, path)), { level: 9 }).length, 0);
  const verdict = total <= BUDGET_BYTES ? 'within' : 'OVER';
  console.log(`Home first-load JS: ${total.toLocaleString('en-AU')} bytes gzipped, ${verdict} the ${BUDGET_BYTES.toLocaleString('en-AU')} byte budget.`);
  if (total > BUDGET_BYTES) process.exitCode = 1;
}

if (process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1]) main();
```

```ts
/** WCAG 2 relative luminance of a #rrggbb colour. */
export function luminance(hex: string): number {
  const m = /^#([0-9a-f]{6})$/i.exec(hex);
  if (!m) throw new Error(`Not a #rrggbb colour: ${hex}`);
  const n = parseInt(m[1], 16);
  const channel = (c: number) => {
    const s = c / 255;
    return s <= 0.03928 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  };
  return 0.2126 * channel((n >> 16) & 255) + 0.7152 * channel((n >> 8) & 255) + 0.0722 * channel(n & 255);
}

/** WCAG 2 contrast ratio between two #rrggbb colours, from 1 to 21. */
export function contrast(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi + 0.05) / (lo + 0.05);
}
```

```ts
/** True when the user's device asks for reduced motion. False on the server. */
export function prefersReducedMotion(): boolean {
  return typeof window !== 'undefined' && typeof window.matchMedia === 'function'
    && window.matchMedia('(prefers-reduced-motion: reduce)').matches;
}

const easeOut = (t: number) => 1 - (1 - t) ** 3;

/** Animates a number with requestAnimationFrame (ease-out). Returns a cancel function. */
export function tween(from: number, to: number, ms: number, onUpdate: (value: number) => void): () => void {
  let start: number | undefined;
  let frame = 0;
  const step = (t: number) => {
    start ??= t;
    const p = Math.min(1, (t - start) / ms);
    onUpdate(p === 1 ? to : from + (to - from) * easeOut(p));
    if (p < 1) frame = requestAnimationFrame(step);
  };
  frame = requestAnimationFrame(step);
  return () => cancelAnimationFrame(frame);
}
```

`components/motion/reveal.tsx` (CSS driven, so it costs almost nothing):

```tsx
'use client';

import { useLayoutEffect, useRef, type ReactNode } from 'react';
import { prefersReducedMotion } from '@/lib/motion';

/**
 * Fades and lifts its content in when it scrolls into view, once. Content is never hidden in the
 * exported HTML: only an element that starts below the first screen is hidden, and only after
 * hydration. `delay` (ms) staggers items in a list, e.g. index * 40.
 */
export function Reveal({ children, delay = 0, className }: { children: ReactNode; delay?: number; className?: string }) {
  const ref = useRef<HTMLDivElement>(null);
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el || typeof IntersectionObserver === 'undefined' || prefersReducedMotion()) return;
    if (el.getBoundingClientRect().top < window.innerHeight) return;
    el.dataset.reveal = 'hidden';
    const io = new IntersectionObserver(([entry]) => {
      if (!entry.isIntersecting) return;
      el.dataset.reveal = 'shown';
      io.disconnect();
    }, { rootMargin: '0px 0px -10% 0px' });
    io.observe(el);
    return () => io.disconnect();
  }, []);
  return <div ref={ref} className={className} style={delay ? { transitionDelay: `${delay}ms` } : undefined}>{children}</div>;
```

```tsx
'use client';

import { useLayoutEffect, useRef, useSyncExternalStore } from 'react';
import { prefersReducedMotion, tween } from '@/lib/motion';

const noop = () => () => {};
/** False while hydrating server HTML, true for anything rendered on the client after that. */
const useAfterHydration = () => useSyncExternalStore(noop, () => true, () => false);

export type CountUpProps = { value: number; format?: (n: number) => string; ms?: number; className?: string };

/**
 * A number that counts up from 0. The final value is in the HTML and always available to screen
 * readers. It animates when it first appears on the client (e.g. after "now" is known) or when it
 * scrolls into view; a value already on screen at hydration stays still, so nothing flashes.
 */
export function CountUp({ value, format = (n) => Math.round(n).toLocaleString('en-AU'), ms = 900, className }: CountUpProps) {
  const ref = useRef<HTMLSpanElement>(null);
  const mountedOnClient = useRef(useAfterHydration()).current;
  useLayoutEffect(() => {
    const el = ref.current;
    if (!el || prefersReducedMotion()) return;
    const run = () => tween(0, value, ms, (v) => { el.textContent = format(v); });
    if (mountedOnClient) return run();
    if (typeof IntersectionObserver === 'undefined' || el.getBoundingClientRect().top < window.innerHeight) return;
    el.textContent = format(0);
    let cancel: (() => void) | undefined;
    const io = new IntersectionObserver(([entry]) => {
      if (!entry.isIntersecting) return;
      io.disconnect();
      cancel = run();
    });
    io.observe(el);
    return () => { io.disconnect(); cancel?.(); };
    // Re-run only when the value changes; format is expected to be stable in meaning.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [value]);
  return (
    <span className={className}>
      <span ref={ref} aria-hidden="true" className="num">{format(value)}</span>
      <span className="sr-only">{format(value)}</span>
    </span>
  );
}
```

```ts
export const TOAST_MS = 4000;
const MAX_TOASTS = 3;

export type ToastAction = { label: string; href: string };
export type Toast = { id: number; message: string; action?: ToastAction };

type ToastState = { toasts: Toast[]; show(message: string, action?: ToastAction): number; dismiss(id: number): void };

let nextId = 1;

export const useToasts = create<ToastState>()((set) => ({
  toasts: [],
  show: (message, action) => {
    const id = nextId++;
    set((s) => ({ toasts: [...s.toasts, { id, message, action }].slice(-MAX_TOASTS) }));

}));

export const toast = (message: string, action?: ToastAction) => useToasts.getState().show(message, action);
```

```ts
/** Fires a burst unless the user turned Celebrations off or prefers reduced motion. */
export async function burst(kind: BurstKind, origin?: Origin): Promise<void> {
  if (!useWorkspace.getState().ws.settings.celebrations || prefersReducedMotion()) return;
  const { default: confetti } = await import('canvas-confetti');

  fired.add(key);
  toast(message, action);
  if (kind) await burst(kind, origin);
```

```ts
import type { Goal, Workspace } from '@/lib/schema';
import { hasDueDate, type DatedGoal } from '@/lib/calendar/events';
import { isDropped } from './applications';

export type PlanProgress = { done: number; total: number; percent: number; next?: DatedGoal };

/**
 * Goals done across applications still being pursued (declined and withdrawn ones are left out),
 * plus goals not tied to an application, and the next open goal by due date.
 */
export function planProgress(ws: Workspace): PlanProgress {
  const dropped = new Set(ws.pipeline.filter(isDropped).map((a) => a.id));
  const goals = ws.goals.filter((g) => !g.applicationId || !dropped.has(g.applicationId));
  const done = goals.filter((g) => g.status === 'done').length;
  const next = goals.filter((g) => g.status !== 'done').filter(hasDueDate)
    .sort((a, b) => a.dueDate.localeCompare(b.dueDate) || a.title.localeCompare(b.title))[0];
  return { done, total: goals.length, percent: goals.length ? Math.round((done / goals.length) * 100) : 0, next };
}

/** The application whose last open goal this status change completes, if any. */
export function completesApplication(goals: readonly Goal[], goalId: string, status: Goal['status']): string | undefined {
  const goal = goals.find((g) => g.id === goalId);
  if (status !== 'done' || !goal?.applicationId || goal.status === 'done') return undefined;
  const siblings = goals.filter((g) => g.applicationId === goal.applicationId && g.id !== goalId);
  return siblings.every((g) => g.status === 'done') ? goal.applicationId : undefined;
}
```
